rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Each signed-in user can only access their own document: users/{their uid}
    match /users/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;

      allow create, update: if request.auth != null
        && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['h', 't', 'ts', 'v'])
        && request.resource.data.h is list
        && request.resource.data.h.size() <= 1000
        && request.resource.data.t is list
        && request.resource.data.t.size() <= 100
        && request.resource.data.ts is number;

      allow delete: if request.auth != null && request.auth.uid == uid;
    }

    // Everything else is blocked
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
